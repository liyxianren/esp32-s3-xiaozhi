完整解决方案
方案概述
在设备端注册定时提醒MCP工具，云端AI调用该工具，设备本地创建FreeRTOS定时器
架构优势
✅ 完全符合小智AI的MCP设计理念
✅ 无需修改云端Python MCP服务器
✅ 不会有TTS朗读指令的问题（正常的MCP调用）
✅ 参数传递准确可靠（结构化JSON，不是文本解析）
✅ 支持所有提醒类型（倒计时、定时、重复）
✅ 断电持久化（可通过NVS存储实现）

void Board::InitializeReminderTools() {
    auto& mcp = McpServer::GetInstance();
    auto& reminder_mgr = Application::GetInstance().GetReminderManager();
    
    // 添加提醒工具
    mcp.AddTool("reminder.add", 
        "Add a timed reminder. Supports countdown, scheduled time, and recurring reminders.\n"
        "Parameters:\n"
        "  delay_seconds: Delay in seconds (for countdown), e.g., '10 seconds later' = 10\n"
        "  message: Reminder message, e.g., 'take medicine'\n"
        "  hour: Hour (for scheduled time), e.g., '7 AM' = 7, default -1 (not used)\n"
        "  minute: Minute (for scheduled time), e.g., '7:30' = 30, default -1 (not used)\n"
        "  repeat: Number of repetitions, default 1 (no repeat), e.g., 'daily reminder' = 10000\n"
        "  interval: Repeat interval in seconds, e.g., 'every hour' = 3600, default 0",
        PropertyList({
            Property("delay_seconds", kPropertyTypeInteger, 0),
            Property("message", kPropertyTypeString),
            Property("hour", kPropertyTypeInteger, -1),
            Property("minute", kPropertyTypeInteger, -1),
            Property("repeat", kPropertyTypeInteger, 1),
            Property("interval", kPropertyTypeInteger, 0)
        }),
        [&reminder_mgr](const PropertyList& props) -> ReturnValue {
            int delay = props["delay_seconds"].value<int>();
            int hour = props["hour"].value<int>();
            int minute = props["minute"].value<int>();
            int repeat = props["repeat"].value<int>();
            int interval = props["interval"].value<int>();
            std::string msg = props["message"].value<std::string>();
            
            // 如果指定了hour或minute，计算实际延迟
            if ((hour >= 0 && hour < 24) || (minute >= 0 && minute < 60)) {
                time_t now;
                time(&now);
                struct tm timeinfo;
                localtime_r(&now, &timeinfo);
                
                if (hour >= 0 && hour < 24) timeinfo.tm_hour = hour;
                if (minute >= 0 && minute < 60) timeinfo.tm_min = minute;
                timeinfo.tm_sec = 0;
                
                time_t target = mktime(&timeinfo);
                if (target <= now) target += 24 * 3600;  // 明天
                
                delay = target - now;
            }
            
            // 创建提醒
            uint32_t id = reminder_mgr.AddReminder(delay, msg);
            
            // 记录用于重复提醒（如果需要）
            if (repeat > 1) {
                // TODO: 实现重复提醒逻辑
                ESP_LOGW("Reminder", "Repeat not yet implemented: repeat=%d, interval=%d", repeat, interval);
            }
            
            // 返回JSON结果
            cJSON* result = cJSON_CreateObject();
            cJSON_AddNumberToObject(result, "id", id);
            cJSON_AddStringToObject(result, "message", msg.c_str());
            cJSON_AddNumberToObject(result, "delay_seconds", delay);
            cJSON_AddStringToObject(result, "status", "created");
            
            return result;
        });
    
    // 取消提醒工具
    mcp.AddTool("reminder.cancel",
        "Cancel a specific reminder by ID",
        PropertyList({
            Property("reminder_id", kPropertyTypeInteger)
        }),
        [&reminder_mgr](const PropertyList& props) -> ReturnValue {
            uint32_t id = props["reminder_id"].value<int>();
            bool success = reminder_mgr.CancelReminder(id);
            return success ? "Reminder cancelled" : "Reminder not found";
        });
    
    // 查询提醒工具
    mcp.AddTool("reminder.list",
        "List all active reminders",
        PropertyList(),
        [&reminder_mgr](const PropertyList& props) -> ReturnValue {
            // TODO: 实现查询功能（需要在ReminderManager添加GetReminders()方法）
            return "Reminder list not yet implemented";
        });
}

void InitializeReminderTools() {
    auto& mcp = McpServer::GetInstance();
    
    // reminder.add 工具
    mcp.AddTool("reminder.add", 
        "添加定时提醒。支持倒计时、定时、重复提醒。\n"
        "参数说明:\n"
        "  delay_seconds: 延迟秒数(倒计时),如'10秒后'则为10\n"
        "  message: 提醒内容,如'吃药'\n"
        "  hour: 小时(定时),如'早上7点'则为7,默认-1表示不使用\n"
        "  minute: 分钟(定时),如'7:30'则为30,默认-1表示不使用\n"
        "  repeat: 重复次数,默认1(不重复),如'每天提醒'则为10000\n"
        "  interval: 重复间隔(秒),如'每小时'则为3600,默认0",
        PropertyList({
            Property("delay_seconds", kPropertyTypeInteger, 0),
            Property("message", kPropertyTypeString),
            Property("hour", kPropertyTypeInteger, -1),
            Property("minute", kPropertyTypeInteger, -1),
            Property("repeat", kPropertyTypeInteger, 1),
            Property("interval", kPropertyTypeInteger, 0)
        }),
        [](const PropertyList& props) -> ReturnValue {
            auto& app = Application::GetInstance();
            auto& reminder_mgr = app.GetReminderManager();
            
            int delay = props["delay_seconds"].value<int>();
            int hour = props["hour"].value<int>();
            int minute = props["minute"].value<int>();
            int repeat = props["repeat"].value<int>();
            int interval = props["interval"].value<int>();
            std::string msg = props["message"].value<std::string>();
            
            ESP_LOGI(TAG, "📥 Received reminder.add: delay=%d, hour=%d, minute=%d, repeat=%d, interval=%d, msg='%s'",
                     delay, hour, minute, repeat, interval, msg.c_str());
            
            // 如果指定了hour或minute，计算到目标时间的延迟
            if ((hour >= 0 && hour < 24) || (minute >= 0 && minute < 60)) {
                time_t now;
                time(&now);
                struct tm timeinfo;
                localtime_r(&now, &timeinfo);
                
                if (hour >= 0 && hour < 24) timeinfo.tm_hour = hour;
                if (minute >= 0 && minute < 60) timeinfo.tm_min = minute;
                timeinfo.tm_sec = 0;
                
                time_t target = mktime(&timeinfo);
                if (target <= now) {
                    target += 24 * 3600;  // 如果时间已过，设为明天
                }
                
                delay = target - now;
                ESP_LOGI(TAG, "⏰ Calculated delay for time-based reminder: %d seconds", delay);
            }
            
            if (delay <= 0) {
                ESP_LOGE(TAG, "❌ Invalid delay: %d", delay);
                throw std::runtime_error("Invalid delay time");
            }
            
            // 创建提醒
            uint32_t id = reminder_mgr.AddReminder(delay, msg);
            ESP_LOGI(TAG, "✅ Reminder created: id=%u, delay=%d, message='%s'", id, delay, msg.c_str());
            
            // TODO: 实现重复提醒
            if (repeat > 1) {
                ESP_LOGW(TAG, "⚠️ Repeat reminders not yet implemented (repeat=%d, interval=%d)", repeat, interval);
            }
            
            // 返回结果
            cJSON* result = cJSON_CreateObject();
            cJSON_AddNumberToObject(result, "id", id);
            cJSON_AddStringToObject(result, "message", msg.c_str());
            cJSON_AddNumberToObject(result, "delay_seconds", delay);
            cJSON_AddNumberToObject(result, "repeat", repeat);
            cJSON_AddNumberToObject(result, "interval", interval);
            
            return result;
        });
    
    // reminder.cancel 工具
    mcp.AddTool("reminder.cancel",
        "取消指定的提醒",
        PropertyList({
            Property("reminder_id", kPropertyTypeInteger)
        }),
        [](const PropertyList& props) -> ReturnValue {
            auto& app = Application::GetInstance();
            auto& reminder_mgr = app.GetReminderManager();
            
            uint32_t id = props["reminder_id"].value<int>();
            ESP_LOGI(TAG, "🗑️ Cancelling reminder: id=%u", id);
            
            bool success = reminder_mgr.CancelReminder(id);
            
            if (success) {
                ESP_LOGI(TAG, "✅ Reminder cancelled: id=%u", id);
                return "Reminder cancelled successfully";
            } else {
                ESP_LOGW(TAG, "⚠️ Reminder not found: id=%u", id);
                return "Reminder not found";
            }
        });
    
    ESP_LOGI(TAG, "✅ Reminder tools initialized");
}

Feature 1: NVS持久化存储
目的：设备断电重启后恢复提醒 实现位置：main/reminder_manager.cc
#include "settings.h"

class ReminderManager {
private:
    Settings settings_;
    
public:
    ReminderManager() : settings_("reminders", true) {
        LoadFromNVS();
    }
    
    ~ReminderManager() {
        SaveToNVS();
    }
    
    void LoadFromNVS() {
        int count = settings_.GetInt("count", 0);
        ESP_LOGI("ReminderMgr", "Loading %d reminders from NVS", count);
        
        for (int i = 0; i < count; i++) {
            std::string prefix = "r" + std::to_string(i) + "_";
            
            uint32_t id = settings_.GetInt(prefix + "id", 0);
            std::string msg = settings_.GetString(prefix + "msg", "");
            time_t trigger_time = settings_.GetInt(prefix + "time", 0);
            
            time_t now;
            time(&now);
            
            if (trigger_time > now) {
                uint32_t delay = trigger_time - now;
                AddReminder(delay, msg);
                ESP_LOGI("ReminderMgr", "Restored reminder: id=%u, delay=%u, msg='%s'", id, delay, msg.c_str());
            }
        }
    }
    
    void SaveToNVS() {
        settings_.SetInt("count", reminders_.size());
        
        int i = 0;
        for (const auto& reminder : reminders_) {
            std::string prefix = "r" + std::to_string(i) + "_";
            settings_.SetInt(prefix + "id", reminder.id);
            settings_.SetString(prefix + "msg", reminder.message);
            settings_.SetInt(prefix + "time", reminder.trigger_time_ms / 1000);
            i++;
        }
        
        ESP_LOGI("ReminderMgr", "Saved %d reminders to NVS", i);
    }
};
Feature 2: 重复提醒
实现位置：main/reminder_manager.h
struct Reminder {
    uint32_t id;
    std::string message;
    uint32_t delay_seconds;
    uint64_t create_time_ms;
    uint64_t trigger_time_ms;
    TimerHandle_t timer;
    
    // ⭐ 新增重复提醒字段
    int repeat_count;       // 剩余重复次数
    int repeat_interval;    // 重复间隔(秒)
};

void ReminderManager::OnTimerExpired(uint32_t id) {
    auto it = std::find_if(reminders_.begin(), reminders_.end(), 
                          [id](const Reminder& r) { return r.id == id; });
    
    if (it != reminders_.end()) {
        // 触发回调
        if (callback_) {
            callback_(it->id, it->message);
        }
        
        // 检查是否需要重复
        if (it->repeat_count > 1) {
            it->repeat_count--;
            
            // 创建新定时器（间隔时间后再次触发）
            TimerContext* ctx = new TimerContext{this, it->id};
            it->timer = xTimerCreate("ReminderTimer",
                                    pdMS_TO_TICKS(it->repeat_interval * 1000),
                                    pdFALSE, ctx, TimerCallback);
            xTimerStart(it->timer, 0);
            
            ESP_LOGI("ReminderMgr", "🔁 Rescheduled reminder #%u (remaining: %d)", 
                     id, it->repeat_count);
        } else {
            // 删除已完成的提醒
            xTimerDelete(it->timer, 0);
            reminders_.erase(it);
            ESP_LOGI("ReminderMgr", "✅ Reminder #%u completed and removed", id);
        }
    }
}

3. FreeRTOS定时器原理
// 创建定时器
TimerContext* ctx = new TimerContext{this, reminder_id};
TimerHandle_t timer = xTimerCreate(
    "ReminderTimer",                          // 定时器名称
    pdMS_TO_TICKS(delay_seconds * 1000),     // 延迟时间(ticks)
    pdFALSE,                                  // 单次触发(不自动重载)
    (void*)ctx,                               // 定时器ID(传递上下文)
    TimerCallback                             // 回调函数
);

// 启动定时器
xTimerStart(timer, 0);

// 静态回调函数
static void TimerCallback(TimerHandle_t timer) {
    TimerContext* ctx = (TimerContext*)pvTimerGetTimerID(timer);
    ctx->manager->OnTimerExpired(ctx->reminder_id);
    delete ctx;  // 释放上下文
}
